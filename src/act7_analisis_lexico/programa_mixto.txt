package main;

import "fmt";

func main() {
    var edad int = 20;
    var opcion int = 2;
    var i int = 0;

    if edad >= 18 {
        fmt.Println("Mayor de edad");
    }

    for i = 0; i < 3; i = i + 1 {
        fmt.Println("Iteracion:", i);
    }

    switch opcion {
        case 1:
            fmt.Println("Opcion uno");

        case:
            fmt.Println("Case incorrecto");

        case 2:
            fmt.Println("Opcion dos");

        default:
            fmt.Println("Otra opcion");
    }

    fmt.Println("Programa mixto terminado");
}

package main;

import "fmt";

func main() {
    var opcion int = 2;
    var i int = 0;

    switch opcion {
        case 1:
            fmt.Println("Opcion uno");

        case:
            fmt.Println("Case sin valor");

        case 3:
            fmt.Println("Opcion tres");

        default:
            fmt.Println("Otra opcion");
    }

    for i = 0; i < 5; {
        fmt.Println("Valor:", i);
    }

    fmt.Println("El analizador continuo");
}

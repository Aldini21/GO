

package main;

import "fmt";

func main() {
    var contador int = 0;
    var i int = 0;

    for contador < 3 {
        fmt.Println("Contador:", contador);

        if contador == 1 {
            fmt.Println("Contador vale uno");
        } else {
            fmt.Println("Contador no vale uno");
        }

        contador = contador + 1;
    }
    
    for {
        i = i + 1;

        if i >= 3 {
            break;
        }

        fmt.Println("Valor de i:", i);
    }

    fmt.Println("Fin del programa");
}

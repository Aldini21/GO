package main;

import "fmt";

func main() {
    var edad int = 20;
    var activo bool = true;

    if edad >= 18 && activo {
        fmt.Println("Usuario mayor y activo");
    } else {
        fmt.Println("Usuario no autorizado");
    }

    var i int = 0;

    for i = 0; i < 3; i = i + 1 {
        fmt.Println("Iteracion:", i);
    }

    switch edad {
        case 18:
            fmt.Println("Tiene dieciocho");

        case 20:
            fmt.Println("Tiene veinte");

        default:
            fmt.Println("Otra edad");
    }
}

package main;
import "fmt";

const MAX_CALIFICACION int = 100;
var escuelaAbierta bool = true;

type SistemaEscolar struct {
    totalAlumnos int;
    sistemaAbierto bool;
}

func registrar() {
    fmt.Println("Registrando...");
}

func calcularPromedio(nota1 int, nota2 int) int {
    var sumaNotas int = nota1 + nota2;
    var promedioFinal int = sumaNotas / 2;

    return promedioFinal;
}

func procesar(limite int) {
    const PASO_MINIMO int = 1;
    var i int = 0;

    for i = 0; i < limite; i++ {
        if i < PASO_MINIMO {
            fmt.Println("Valor pequeno:", i);
        } else {
            fmt.Println("Valor:", i);
        }
    }
}

func main() {
    var resultado int = 0;
    var nombreAlumno string = "Ana";

    registrar();
    resultado = calcularPromedio(90, MAX_CALIFICACION);
    fmt.Println(nombreAlumno, resultado);

    procesar(3);
}

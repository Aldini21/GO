import "fmt";

/* Actividad 12: programa valido (codigo intermedio de operaciones) */

const LIMITE int = 20;

func main() {
    var x int = 5;
    var y int = 10;
    var z int = x + y;
    var w int = 2;
    var a int = 3;
    var b int = 4;
    var c int = 6;
    var d int = 1;

    var resultado int = a * (b + c);
    var flag bool = (x > y) && (z < LIMITE);
    var total int = (a + b) * (c - d);
    resultado = (x + y) / (z - w);
    resultado = a / b;
    flag = !flag || total >= 30;
    fmt.Println(resultado, total, flag);
}
